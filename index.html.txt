<script>
let lastFetchedId = "";
let currentMode = "1M";
const B_POOL = [5,6,7,8,9], S_POOL = [0,1,2,3,4];

// API Endpoints fix for 30s
const API_ENDPOINTS = {
    '30S': 'https://draw.ar-lottery01.com/WinGo/WinGo_30/GetHistoryIssuePage.json',
    '1M': 'https://draw.ar-lottery01.com/WinGo/WinGo_1M/GetHistoryIssuePage.json',
    '3M': 'https://draw.ar-lottery01.com/WinGo/WinGo_3M/GetHistoryIssuePage.json',
    '5M': 'https://draw.ar-lottery01.com/WinGo/WinGo_5M/GetHistoryIssuePage.json'
};

// Smooth Interactivity with High Inertia
interact('#hack-popup').draggable({
  inertia: {
    resistance: 20,
    minSpeed: 150,
    endSpeed: 50
  },
  modifiers: [
    interact.modifiers.restrictRect({
      restriction: 'parent',
      endOnly: true
    })
  ],
  listeners: {
    move (event) {
      const target = event.target;
      const x = (parseFloat(target.getAttribute('data-x')) || 0) + event.dx;
      const y = (parseFloat(target.getAttribute('data-y')) || 0) + event.dy;
      target.style.transform = `translate(calc(-50% + ${x}px), ${y}px)`;
      target.setAttribute('data-x', x);
      target.setAttribute('data-y', y);
    }
  }
});

function toggleHack() {
  const popup = document.getElementById('hack-popup');
  const showBtn = document.getElementById('show-btn');
  if (popup.classList.contains('hidden')) {
    popup.classList.remove('hidden');
    showBtn.style.display = 'none';
  } else {
    popup.classList.add('hidden');
    showBtn.style.display = 'block';
  }
}

function setMode(mode, element) {
    currentMode = mode;
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    element.classList.add('active');
    document.getElementById('period-val').value = "FETCHING...";
    document.getElementById('number-text').value = "MODE: " + mode;
    lastFetchedId = ""; 
    updatePeriod();
}

async function fetchGameData() {
    try {
        const response = await fetch(API_ENDPOINTS[currentMode], { cache: "no-store" });
        const data = await response.json();
        return data.data.list;
    } catch (e) { 
        console.error("Fetch Error:", e);
        return null; 
    }
}

async function updatePeriod() {
  const list = await fetchGameData();
  if(list && list.length > 0) {
      const latestIssue = list[0].issueNumber;
      const nextIssueFull = (BigInt(latestIssue) + 1n).toString();
      document.getElementById('period-val').value = nextIssueFull;
      lastFetchedId = latestIssue;
  }
}

async function generateResult() {
  const btn = document.getElementById('analysis-btn');
  const predText = document.getElementById('prediction-text');
  const numText = document.getElementById('number-text');
  
  btn.disabled = true;
  predText.value = "LOADING...";
  
  const list = await fetchGameData();
  
  setTimeout(() => {
    if (list && list.length > 0) {
      const last5 = list.slice(0, 5).map(x => parseInt(x.number) >= 5 ? "BIG" : "SMALL");
      const nextPred = last5.filter(x => x === "BIG").length > 2 ? "BIG" : "SMALL";
      const pool = nextPred === "BIG" ? S_POOL : B_POOL;
      const opposites = pool.sort(() => 0.5 - Math.random()).slice(0, 2);
      
      predText.value = nextPred;
      numText.value = "TARGET: " + opposites.join(", ");
      document.getElementById('server-status').textContent = currentMode + " CONNECTED";
      document.getElementById('server-status').style.color = "var(--neon-green)";
    } else {
      predText.value = "TIMEOUT";
      numText.value = "Check API Link";
    }
    btn.disabled = false;
  }, 800);
}

// Frequent updates for 30s accuracy
setInterval(updatePeriod, 2000);
updatePeriod();
